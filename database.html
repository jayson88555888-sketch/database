💻 Basic SQL Examples

Common SQL commands used with databases.

SQL Example 
-- Create a table

CREATE TABLE Students (
    StudentID INT PRIMARY KEY,
    Name VARCHAR(50),
    Course VARCHAR(50),
    Age INT
);


-- Insert data

INSERT INTO Students
(StudentID, Name, Course, Age)
VALUES
(1, 'Alice', 'IT', 18);


-- Display data

SELECT *
FROM Students;


-- Update data

UPDATE Students
SET Age = 19
WHERE StudentID = 1;


-- Delete data

DELETE FROM Students
WHERE StudentID = 1;
